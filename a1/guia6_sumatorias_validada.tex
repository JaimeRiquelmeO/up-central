\section*{Resumen teórico}

\begin{theorybox}{1. Propiedades de las sumatorias}
\begin{enumerate}[label=\arabic*.,itemsep=0.55em]
  \item $\displaystyle \sum_{k=m}^{n}a_k=0$ cuando $m>n$ (convención de suma vacía).
  \item $\displaystyle \sum_{k=m}^{n}c\,a_k=c\sum_{k=m}^{n}a_k$, si $c$ no depende de $k$.
  \item $\displaystyle \sum_{k=m}^{n}c=c(n-m+1)$, para $m\le n$.
  \item $\displaystyle \sum_{k=m}^{n}(a_k+b_k)=\sum_{k=m}^{n}a_k+\sum_{k=m}^{n}b_k$.
  \item El índice es mudo: $\displaystyle \sum_{k=m}^{n}a_k=\sum_{j=m}^{n}a_j$.
  \item Traslación: $\displaystyle \sum_{k=m}^{n}a_k=\sum_{k=m+r}^{n+r}a_{k-r}$.
  \item Separación: $\displaystyle \sum_{k=m}^{n}a_k=\sum_{k=m}^{r}a_k+\sum_{k=r+1}^{n}a_k$, si $m\le r\le n$.
  \item $\displaystyle \sum_{k=m}^{m}a_k=a_m$.
  \item Telescópica: $\displaystyle \sum_{k=m}^{n}(a_{k+1}-a_k)=a_{n+1}-a_m$.
\end{enumerate}
\end{theorybox}

\begin{theorybox}{2. Sumatorias notables}
\begin{align*}
\sum_{k=1}^{n}k&=\frac{n(n+1)}2,&
\sum_{k=1}^{n}k^2&=\frac{n(n+1)(2n+1)}6,\\
\sum_{k=1}^{n}k^3&=\left(\frac{n(n+1)}2\right)^2,&
\sum_{k=1}^{n}ar^{k-1}&=a\frac{1-r^n}{1-r}\quad(r\ne1).
\end{align*}
\end{theorybox}

\section*{I. Ejercicios propuestos}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=*,itemsep=1.1em]
\item Si $\sum_{i=1}^{10}x_i^2=196$, $\sum_{i=1}^{10}(2x_i-3)^2=1130$,
$x_8=x_9=-3$ y $x_{10}=5$, determine $\displaystyle\sum_{i=1}^{7}x_i$.

\item Determine $b\in\R$ tal que
\[\sum_{k=2}^{8}\frac{2b-4}{(5k-9)(5k+1)}=\frac{329}{1476}.\]

\item Encuentre los $p<0$ para los cuales
\[\sum_{k=2}^{p+2}\frac{k-2p}{2}=\sum_{k=4}^{p+4}(3p+1).\]
\textit{Nota de dominio:} los límites de una sumatoria finita deben ser enteros; por ello se entiende $p\in\Z$.

\item Determine $n\in\N$ tal que
\[\sum_{k=n-1}^{2n-2}\left(\frac{k}{3}+\frac12\right)
=\sum_{k=3}^{11}\frac{115}{8k^2-2}.\]

\item Calcule $\displaystyle S=\sum_{i=10}^{40}i(i+1)^2$.

\item Sea
\[a_k=\begin{cases}(1/3)^k,&1\le k<100,\\(k+1)^2,&100\le k\le200.
\end{cases}\]
Calcule $\displaystyle S=\sum_{k=1}^{200}a_k$.

\item Demuestre que
\[\sum_{i=1}^{n}(3i-2)=\frac{n(3n-1)}2\qquad(n\in\N).\]

\item Demuestre que
\[\sum_{i=0}^{n}a^i=\frac{a^{n+1}-1}{a-1}\qquad(a\ne1),\]
y úselo para calcular $\displaystyle S=\sum_{i=1}^{100}(1/2)^i$.

\item Considere
\[a_n=\begin{cases}n^2+2n,&1\le n\le8,\\
\sqrt[3]{3n}-\sqrt[3]{3n+3},&n\ge9.
\end{cases}\]
Determine $\displaystyle\sum_{k=2}^{71}a_k$.

\item Determine los valores admisibles de $c$ en
\[\sum_{k=1}^{20}(k^2-10)-\sum_{k=1}^{c}(k+9)=2610.\]
\textit{Nota de dominio:} se entiende $c\in\N_0$, pues es el límite superior de una suma finita.

\item Si $\displaystyle\sum_{k=1}^{n}a_k=2n^2+3n$, calcule
$\displaystyle\sum_{k=7}^{20}\frac{a_k-5}{2}$.

\item Si
\[\sum_{k=1}^{8}a_k=120,\qquad \sum_{k=1}^{8}a_k^2=160,
\qquad a_9=6,\qquad a_{10}=8,\]
calcule:
\begin{enumerate}[label=\roman*),itemsep=0.25em]
  \item $\displaystyle\sum_{k=1}^{10}a_k^2$;
  \item $\displaystyle\sum_{k=1}^{9}a_k(a_k-2)$;
  \item $\displaystyle\sum_{k=1}^{10}(a_k-1)^2-\sum_{k=1}^{8}(a_k-1)^2$.
\end{enumerate}

\item Calcule $\displaystyle\sum_{k=16}^{80}\frac1{\sqrt{k}+\sqrt{k+1}}$.
\end{enumerate}
